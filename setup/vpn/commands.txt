# установка (кладёт vpn и no-vpn в /usr/local/bin, гасит старый systemd-юнит)
./install.sh

# управление неймспейсом
vpn --ensure          # поднять сеть в ns (идемпотентно, вызывается автоматически)
vpn --status          # что сейчас настроено
vpn --check           # внешний IP: в текущем ns и через vpn
vpn --clean           # снести ns, вернуть ens36 на хост

# порты хостовой петли, видимые изнутри ns
# автоматически: порты IDE из ~/.claude/ide/<порт>.lock
# дополнительно:  номера портов в ~/.no-vpn-ports (пробелы/запятые/строки, # — комментарий)
echo 15174 >> ~/.no-vpn-ports && vpn --ensure   # применить (пересинхронизация на каждом запуске)

# хосты, куда из ns ходим мимо vpn (через хост)
# ~/.no-vpn-hosts: имена, IPv4-адреса, IPv4-сети a.b.c.d/len (пробелы/запятые/строки, # — комментарий)
# имена резолвятся во все IPv4-адреса хостовым DNS и DNS ns, на каждом запуске vpn
echo school.mos.ru >> ~/.no-vpn-hosts && vpn --ensure
sudo ip netns exec vpn ip route show proto 222   # маршруты обхода в ns

# запуск команд
vpn bash
MY_VAR=123 vpn bash
vpn curl -s https://ifconfig.me ; echo
no-vpn <команда>      # выполнить в основном ns хоста (изнутри vpn/socks)

# ручные проверки
sudo ip netns exec vpn ip addr show ens36
sudo ip netns exec vpn ip route show
sudo ip netns exec vpn cat /etc/resolv.conf
sudo ip netns exec vpn ping -c 3 google.com
sudo ip netns exec vpn sudo -u $USER bash
